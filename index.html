// --- Blok 3: LRCLIB API Integratie (Ondersteunt Spotify URLs en Tekst) --- async function fetchLyricsDirect() { const query = document.getElementById('lyricsQuery').value.trim(); const resultBox = document.getElementById('lyricsResult'); const btn = document.getElementById('lyricsBtn'); if (!query) { alert('Voer een artiest/nummer in of plak een Spotify link.'); return; } btn.disabled = true; resultBox.innerHTML = `⏳ Zoeken... Songtekst wordt opgehaald.`; try { let url = ''; const match = query.match(/track\/([a-zA-Z0-9]+)/); if (match && match[1]) { // Als het een Spotify link is, zoek op Spotify Track ID url = `https://lrclib.net/api/get-by-spotify-id/${match[1]}`; } else { // Anders gewoon zoeken op tekst url = `https://lrclib.net/api/search?q=${encodeURIComponent(query)}`; } const response = await fetch(url); if (!response.ok) { throw new Error("Niet gevonden"); } const data = await response.json(); // Verwerk resultaat (get-by-spotify-id geeft 1 object terug, search een array) const item = Array.isArray(data) ? data[0] : data; if (item && (item.syncedLyrics || item.plainLyrics)) { const lyrics = item.syncedLyrics || item.plainLyrics; resultBox.innerText = `🎵 ${item.artistName} - ${item.trackName}\n\n${lyrics}`; } else { resultBox.innerHTML = `Niet gevonden: Geen songtekst aangetroffen in de database. Probeer een zoekterm zoals "Artiest - Nummer".`; } } catch (err) { resultBox.innerHTML = `Niet gevonden: Geen match voor deze Spotify-link of zoekopdracht. Probeer een gewone zoekterm (bijv. Flemming - Automatisch).`; } finally { btn.disabled = false; } }
